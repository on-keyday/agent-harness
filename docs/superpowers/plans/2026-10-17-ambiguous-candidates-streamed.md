# Ambiguous-runner candidates streamed — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** When a spawn matches more than one (runner, agent) choice, the server answers with a per-runner candidate list on a trsf stream, under one rule for interactive and oneshot, and every surface lets the operator pick and re-issue pinned by runner identity.

**Architecture:** `OpenInteractiveResponse` and `SubmitResponse` carry only a `candidates_stream_id` in their `ambiguous_runner` branch; the rows travel as `RunnerCandidatesBody` through the board change's `respondWithBody` / `cli.ReadResponseBody`. The server builds that body from the existing combo expansion, grouped by runner. Clients get one `cli.AmbiguousRunnerError` from both paths, and a candidate's `Pin()` decides the selector.

**Tech Stack:** Go (server, cli, tui with bubbletea, wasm bridge), `.bgn` schema regenerated by `make protoregen` (ebm2go), plain JS in `webui/static/main.js` tested by `node --test` (`make js-test`), Python scripts.

**Spec:** `docs/superpowers/specs/2026-10-17-ambiguous-candidates-streamed-design.md`. Read it before any task; section numbers below (§1–§6) refer to it.

## Review Focus

1. **A runner advertising no profiles (legacy) or an empty-name profile.** `advertisedProfiles` returns `[AgentBin]` or `[""]`; a candidate row must still be pickable and re-issue with `--agent ""` meaning "runner default". Pinned in Task 1's grouping test.
2. **Two candidates that are the SAME runner identity (takeover window).** `Registry.Add` replaces the identity's connection but the displaced entry may still be listed; grouping is by connection id, so two rows can share `runner_id`. `Pin()` then pins by identity and the scheduler binds to whichever connection holds it — acceptable, but the grouping must not merge them or drop one. Pinned in Task 1.
3. **A candidate whose identity is zero.** `Pin()` must fall back to the cid, and the wasm/JS path must not send an empty `runner`. Pinned in Task 3 and Task 5.
4. **Picker re-issue of a oneshot that then QUEUES.** The pin must survive a runner reconnect (ByRunnerId), which is the whole reason for D4. Pinned by the scheduler test in Task 3.
5. **The TUI picker opened with exactly one runner and one profile** (cannot be ambiguous by construction, but a resumed or legacy shape could deliver it): both steps skip and the pick fires immediately rather than showing an empty step. Pinned in Task 4.

```go
package server

import "github.com/on-keyday/agent-harness/runner/protocol"

// candidatesBody turns the combos of an ambiguous spawn into one row per
// runner connection, in the order the combos arrive (Registry.Candidates'
// order), each row listing the profiles offered on it. Grouping is by
// connection, not identity: during a reconnect takeover two connections can
// briefly share one identity, and each is a separate row.
func candidatesBody(combos []interactiveCombo, repo string) *protocol.RunnerCandidatesBody {
	var rows []protocol.RunnerCandidate
	index := map[string]int{}
	for _, c := range combos {
		key := c.Entry.ID.String()
		i, ok := index[key]
		if !ok {
			var rc protocol.RunnerCandidate
			rc.RunnerId = c.Entry.Identity
			rc.SetCid([]byte(key))
			rc.SetHostname([]byte(c.Entry.Hostname))
			rc.SetMatchedRoot([]byte(matchedRoot(c.Entry.AllowedRoots, repo)))
			rc.ActiveTasks = uint16(len(c.Entry.ActiveTasks))
			rc.MaxTasks = uint16(c.Entry.MaxTasks)
			rows = append(rows, rc)
			i = len(rows) - 1
			index[key] = i
		}
		var name protocol.AgentProfileName
		name.SetName([]byte(c.Profile))
		rows[i].SetProfiles(append(rows[i].Profiles, name))
	}
	var body protocol.RunnerCandidatesBody
	body.SetCandidates(rows)
	return &body
}
```

// RunnerCandidate is one runner offered when a spawn was ambiguous. Profiles
// are the agents the caller may pick on it. RunnerID is the 32-hex identity,
// empty for a runner that predates identity.
type RunnerCandidate struct {
	RunnerID    string
	Cid         string
	Hostname    string
	MatchedRoot string
	ActiveTasks int
	MaxTasks    int
	Profiles    []string
}

// RunnerCombo is one (runner, profile) choice out of a candidate list.
type RunnerCombo struct {
	Candidate RunnerCandidate
	Profile   string
}

// FlattenCandidates lists every (runner, profile) choice in candidate order.
func FlattenCandidates(cs []RunnerCandidate) []RunnerCombo {
	var out []RunnerCombo
	for _, c := range cs {
		for _, p := range c.Profiles {
			out = append(out, RunnerCombo{Candidate: c, Profile: p})
		}
	}
	return out
}

func candidatesFromBody(body *protocol.RunnerCandidatesBody) []RunnerCandidate {
	out := make([]RunnerCandidate, 0, len(body.Candidates))
	for _, c := range body.Candidates {
		rc := RunnerCandidate{
			Cid: string(c.Cid), Hostname: string(c.Hostname), MatchedRoot: string(c.MatchedRoot),
			ActiveTasks: int(c.ActiveTasks), MaxTasks: int(c.MaxTasks),
		}
		if !c.RunnerId.IsZero() {
			rc.RunnerID = c.RunnerId.Hex()
		}
		for _, p := range c.Profiles {
			rc.Profiles = append(rc.Profiles, string(p.Name))
		}
		out = append(out, rc)
	}
	return out
}

// ambiguous answer — interactive and submit, native and wasm — goes through

// formatAmbiguousCandidates renders an ambiguous spawn's candidates for the
// CLI: one line per runner with the flags that pin it. With one runner the
// --runner flag is noise and is left out; --agent is shown whenever the row
// offers more than one agent.
func formatAmbiguousCandidates(cands []cli.RunnerCandidate) string {
	var b strings.Builder
	if len(cands) == 1 {
		fmt.Fprintf(&b, "ambiguous agent: %d agents on this runner; re-run with --agent <name>:\n", len(cands[0].Profiles))
	} else {
		fmt.Fprintf(&b, "ambiguous runner: %d runners match this repo; re-run with the flags shown:\n", len(cands))
	}
	for _, c := range cands {
		var hint []string
		if len(cands) > 1 {
			hint = append(hint, "--runner "+c.Pin().Runner)
		}
		agents := make([]string, len(c.Profiles))
		for i, p := range c.Profiles {
			if p == "" {
				p = "(default)"
			}
			agents[i] = p
		}
		if len(c.Profiles) > 1 || (len(c.Profiles) == 1 && c.Profiles[0] != "") {
			hint = append(hint, "--agent <"+strings.Join(agents, "|")+">")
		}
		fmt.Fprintf(&b, "  %-18s [%d/%d]  %s  %s\n", c.Hostname, c.ActiveTasks, c.MaxTasks, c.MatchedRoot, strings.Join(hint, " "))
	}
	return b.String()
}

// pickerRetry re-issues the request that came back ambiguous, pinned to what
// the operator picked. It travels with the failing request's own result
// message, so a picker can never re-issue some other request.
type pickerRetry func(sel cli.SelectorOpts, agent string) tea.Cmd

// RunnerPickerModel is shown when a spawn comes back ambiguous_runner. Step 1
// picks a runner, step 2 an agent on it; a step with one choice is skipped.
// Digits pick directly, j/k + Enter reach any row.
type RunnerPickerModel struct {
	open   bool
	cands  []cli.RunnerCandidate
	retry  pickerRetry
	runner int // -1 while in step 1
	cursor int
}

func (p *RunnerPickerModel) IsOpen() bool { return p.open }

func (p *RunnerPickerModel) Close() { *p = RunnerPickerModel{} }

// Open starts a pick. It returns the retry's Cmd straight away when there is
// nothing to choose.
func (p *RunnerPickerModel) Open(cands []cli.RunnerCandidate, retry pickerRetry) tea.Cmd {
	*p = RunnerPickerModel{open: true, cands: cands, retry: retry, runner: -1}
	if len(cands) == 1 {
		return p.chooseRunner(0)
	}
	return nil
}

func (p *RunnerPickerModel) rows() int {
	if p.runner < 0 {
		return len(p.cands)
	}
	return len(p.cands[p.runner].Profiles)
}

func (p *RunnerPickerModel) chooseRunner(i int) tea.Cmd {
	p.runner, p.cursor = i, 0
	if len(p.cands[i].Profiles) == 1 {
		return p.chooseAgent(0)
	}
	return nil
}

func (p *RunnerPickerModel) chooseAgent(i int) tea.Cmd {
	c := p.cands[p.runner]
	cmd := p.retry(c.Pin(), c.Profiles[i])
	p.Close()
	return cmd
}

func (p *RunnerPickerModel) choose(i int) tea.Cmd {
	if p.runner < 0 {
		return p.chooseRunner(i)
	}
	return p.chooseAgent(i)
}

// Key handles one key while open. The bool reports whether a request was
// re-issued (the caller logs the pick).
func (p *RunnerPickerModel) Key(key string) (tea.Cmd, bool) {
	switch key {
	case "esc":
		if p.runner >= 0 && len(p.cands) > 1 {
			p.runner, p.cursor = -1, 0
			return nil, false
		}
		p.Close()
		return nil, false
	case "j", "down":
		if p.cursor < p.rows()-1 {
			p.cursor++
		}
		return nil, false
	case "k", "up":
		if p.cursor > 0 {
			p.cursor--
		}
		return nil, false
	case "enter":
		cmd := p.choose(p.cursor)
		return cmd, !p.open
	}
	if len(key) == 1 && key[0] >= '1' && key[0] <= '9' {
		if i := int(key[0] - '1'); i < p.rows() {
			cmd := p.choose(i)
			return cmd, !p.open
		}
	}
	return nil, false
}

func (p *RunnerPickerModel) View() string {
	if !p.open {
		return ""
	}
	box := lipgloss.NewStyle().Border(lipgloss.RoundedBorder()).BorderForeground(colorFocused).Padding(1, 2)
	var b strings.Builder
	mark := func(i int) string {
		if i == p.cursor {
			return "›"
		}
		return " "
	}
	if p.runner < 0 {
		b.WriteString("Ambiguous runner — pick one:\n\n")
		for i, c := range p.cands {
			fmt.Fprintf(&b, "%s %d) %-16s %s  [%d/%d]  %s\n", mark(i), i+1, c.Hostname, c.MatchedRoot,
				c.ActiveTasks, c.MaxTasks, strings.Join(c.Profiles, " "))
		}
	} else {
		c := p.cands[p.runner]
		fmt.Fprintf(&b, "Agent on %s — pick one:\n\n", c.Hostname)
		for i, a := range c.Profiles {
			if a == "" {
				a = "(default)"
			}
			fmt.Fprintf(&b, "%s %d) %s\n", mark(i), i+1, a)
		}
	}
	b.WriteString("\n" + FooterStyle.Render("number or j/k+Enter to pick · Esc back/cancel"))
	return box.Render(b.String())
}
