# Task exec Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Run a command in a task's worktree as its own process — stdout and stderr separated, the command's exit code returned — from the CLI, the TUI, the WebUI and the ssh gateway.

**Architecture:** A client asks the server to start one; the server allocates an id, creates a data stream (spliced client↔runner) and a control stream, and registers it. The runner runs the argv with `agentexec.ExecuteCommandWithOption(…, ptyEnabled=false, …)` in `<repo>/.harness-worktrees/<task-id>/` and reports the outcome; the server pushes one `ExecEvent` onto the control stream. No task is created; `exec ls` / `exec kill` read and act on the registry, and `TaskInfo.exec_count` puts the fact on the task's own row.

**Tech Stack:** Go 1.25.7, `runner/protocol` (`.bgn` → `scripts/protoregen.sh`), `github.com/on-keyday/objtrsf/exec` and `/trsf`.

**Spec:** `docs/superpowers/specs/2026-08-25-task-exec-design.md` — read the **Problem** section verbatim before starting, and report which of P1–P3 your diff addresses.

- **This is a `.bgn` change.** Before landing: `scripts/wire-skew-check.sh`. When deploying: **restart the server FIRST** — an old server cannot decode a new runner's messages, and that has killed the whole runner fleet before (Pitfall 10).
- **The whole schema lands in Task 1.** No "also add this field in Task N" (`feedback_no_split_schemas`).
- Regenerate with `scripts/protoregen.sh` (no arguments regenerates `runner/protocol/message.bgn`). Never hand-edit `runner/protocol/message.go`.
- Verify with `make check`, `make wasm-check`, `make vet`, `make test`, `make test-integration` — not ad-hoc `go build ./...`.
- **Never** run bare `go build ./cmd/<x>` — it drops a binary in the worktree root. Use `go build -o /dev/null ./cmd/<x>`.
- The gate for running an exec is **the worktree directory existing**, never the task's status (D5). A task that ended dirty keeps its worktree and is a valid target.
- An exec dies with its caller (D10). There is no detached form; `submit` is that feature.
- `exec_count` is printed at zero and omitted only when the subject does not exist (D11) — the observer-count rule.

# One argv element. u16 per element matches ClaudeArg: individual arguments are
# short in practice and 64 KiB is past any real one.
format ExecArg:
    arg_len :u16
    arg :[arg_len]u8

format ExecArgv:
    argv_len :u16
    argv :[argv_len]ExecArg

format ExecRunRequest:
    task_id :TaskID
    argv :ExecArgv
    # stdin_enabled=0 closes the child's stdin immediately, which is what a
    # non-interactive caller wants: a child that reads stdin gets EOF rather
    # than hanging on a pipe nobody will write to.
    stdin_enabled :u1
    reserved :u7

enum ExecRunStatus:
    :u8
    ok
    not_found            # unknown task id, or outside the caller's scope
    no_worktree          # no worktree directory to run in — never had one, or
                         # it was removed at a clean task end
    runner_unreachable
    empty_argv
    denied               # capability check failed
    internal_error

format ExecRunResponse:
    status :ExecRunStatus
    exec_id :u64
    data_stream_id :u64
    control_stream_id :u64

enum ExecEventKind:
    :u8
    exited               # the child ran and ended; exit_code is its own
    killed               # exec_run_kill, or the runner was torn down
    failed               # the child never started (no such binary, cwd gone)

format ExecEvent:
    kind :ExecEventKind
    # The child's own exit code for `exited`, -1 for the other two — the
    # convention TaskFinished uses, for the same reason: a signal death has no
    # exit code to report.
    exit_code :i32
    detail :[..]u8       # human-readable; empty for a clean `exited`

format RunnerExecRunRequest:
    exec_id :u64
    task_id :TaskID
    auth_ticket :[16]u8
    stream_id :u64
    # The repo path travels from the server's task record, as
    # RunnerGitQueryRequest's does and for the same reason: the runner drops its
    # per-task entry when a task ends, so it cannot resolve the worktree for a
    # TERMINAL task on its own. The runner re-validates this against --roots.
    repo_path_len :u16
    repo_path :[repo_path_len]u8
    argv :ExecArgv
    stdin_enabled :u1
    reserved :u7

format ExecRunFinished:
    exec_id :u64
    exit_code :i32
    kind :ExecEventKind
    detail :[..]u8

format ExecRunInfo:

package protocol

import "testing"

// Every new format round-trips, and argv survives element boundaries — the
// length-prefixed-list shape is where an off-by-one shows up as a silently
// truncated command rather than an error.
func TestExecRunRequestRoundTrip(t *testing.T) {
	var argv ExecArgv
	for _, a := range []string{"sh", "-c", "echo hi && echo bye 1>&2"} {
		var e ExecArg
		if !e.SetArg([]byte(a)) {
			t.Fatalf("SetArg(%q) failed", a)
		}
		argv.Argv = append(argv.Argv, e)
	}
	argv.ArgvLen = uint16(len(argv.Argv))

	in := ExecRunRequest{TaskId: TaskID{Id: [16]byte{1, 2, 3}}, Argv: argv}
	in.SetStdinEnabled(true)

	buf, err := in.Append(nil)
	if err != nil {
		t.Fatalf("Append: %v", err)
	}
	var out ExecRunRequest
	if err := out.DecodeExact(buf); err != nil {
		t.Fatalf("DecodeExact: %v", err)
	}
	if out.Argv.ArgvLen != 3 {
		t.Fatalf("argv len = %d, want 3", out.Argv.ArgvLen)
	}
	if got := string(out.Argv.Argv[2].Arg); got != "echo hi && echo bye 1>&2" {
		t.Errorf("argv[2] = %q, want the third argument intact", got)
	}
	if !out.StdinEnabled() {
		t.Error("stdin_enabled did not survive")
	}
}

func TestExecEventRoundTrip(t *testing.T) {
	in := ExecEvent{Kind: ExecEventKind_Exited, ExitCode: 3}
	buf, err := in.Append(nil)
	if err != nil {
		t.Fatalf("Append: %v", err)
	}
	var out ExecEvent
	if err := out.DecodeExact(buf); err != nil {
		t.Fatalf("DecodeExact: %v", err)
	}
	if out.Kind != ExecEventKind_Exited || out.ExitCode != 3 {
		t.Errorf("got kind=%v code=%d, want exited/3", out.Kind, out.ExitCode)
	}
	// A negative code must survive: -1 is what killed/failed report, and an
	// unsigned field would turn it into 4294967295.
	in2 := ExecEvent{Kind: ExecEventKind_Killed, ExitCode: -1}
	buf2, _ := in2.Append(nil)
	var out2 ExecEvent
	if err := out2.DecodeExact(buf2); err != nil {
		t.Fatalf("DecodeExact: %v", err)
	}
	if out2.ExitCode != -1 {
		t.Errorf("exit code = %d, want -1", out2.ExitCode)
	}
}

// `all` must cover the new bit, and the new bit must not collide.
func TestExecRunCapabilityBit(t *testing.T) {
	if Capability_ExecRun&Capability_All != Capability_ExecRun {
		t.Error("Capability_All does not include exec_run")
	}
	for _, other := range []Capability{
		Capability_Spawn, Capability_Cancel, Capability_ExecControl,

		Capability_ForwardRemote, Capability_Notify, Capability_Prune,

```go
package runner

import (
	"context"
	"fmt"
	"os"
	"path/filepath"

	"github.com/on-keyday/agent-harness/appwire"
	"github.com/on-keyday/agent-harness/peer"
	"github.com/on-keyday/agent-harness/runner/protocol"
	agentexec "github.com/on-keyday/objtrsf/exec"
	"github.com/on-keyday/objtrsf/trsf"
)

// execRunDir is where an exec runs: the task's worktree.
func execRunDir(repoPath, taskIDHex string) string {
	return filepath.Join(repoPath, ".harness-worktrees", taskIDHex)
}

// execRunDirExists reports whether that directory is there.
//
// This is the whole gate (spec D5). NOT the task's status: a task that ended
// with uncommitted work keeps its worktree — RemoveIfClean declines to remove a
// dirty one — and running `git status` or `make test` in that tree is exactly
// what an operator wants then. A task that ended clean had it removed and has
// nowhere to run.
func execRunDirExists(repoPath, taskIDHex string) bool {
	st, err := os.Stat(execRunDir(repoPath, taskIDHex))
	return err == nil && st.IsDir()
}

// handleExecRun runs one argv in a task's worktree and reports the outcome.
//
// The data plane is the same call the event-stream task kind makes
// (runner/streamtask.go): ptyEnabled=false, so stdout and stderr arrive as
// separate frame types instead of interleaved on one PTY.
func (s *Session) handleExecRun(ctx context.Context, req *protocol.RunnerExecRunRequest) {
	log := s.logger()
	taskIDHex := hexTaskID(req.TaskId)
	repoPath := string(req.RepoPath)

	finish := func(kind protocol.ExecEventKind, code int32, detail string) {
		m := &protocol.RunnerMessage{Kind: protocol.RunnerMessageType_ExecRunFinished}
		body := protocol.ExecRunFinished{ExecId: req.ExecId, ExitCode: code, Kind: kind}
		body.Detail = []byte(detail)
		m.SetExecRunFinished(body)
		data := m.MustAppend([]byte{byte(appwire.AppKind_RunnerControl)})
		_ = s.Sender.Send(data)
	}

	stream := peer.WaitForBidirectionalStream(ctx, s.Streams, trsf.StreamID(req.StreamId))
	if stream == nil {
		log.Error("exec_run: stream not visible", "stream_id", req.StreamId)
		finish(protocol.ExecEventKind_Failed, -1, "exec stream not visible")
		return
	}

	// The repo path came from the server's task record, so re-validate it here
	// the way handleGitQuery does: the runner decides what it will run in.
	if !s.repoAllowed(repoPath) {
		_ = stream.CloseBoth()
		finish(protocol.ExecEventKind_Failed, -1, "repo not allowed by --roots: "+repoPath)
		return
	}
	if !execRunDirExists(repoPath, taskIDHex) {
		_ = stream.CloseBoth()
		finish(protocol.ExecEventKind_Failed, -1, "no worktree for task "+taskIDHex)
		return
	}
	if req.Argv.ArgvLen == 0 {
		_ = stream.CloseBoth()
		finish(protocol.ExecEventKind_Failed, -1, "empty argv")
		return
	}

	argv := make([]string, 0, req.Argv.ArgvLen)
	for i := range req.Argv.Argv {
		argv = append(argv, string(req.Argv.Argv[i].Arg))
	}
	dir := execRunDir(repoPath, taskIDHex)

	var exitCode int32 = -1
	kind := protocol.ExecEventKind_Failed
	var detail string
	runErr := agentexec.ExecuteCommandWithOption(ctx, stream, log,
		argv[0], argv[1:], dir,
		false, // no PTY: separate stdout and stderr is the point
		s.execRunEnv(taskIDHex),
		agentexec.ExecuteOption{
			OnProcessExit: func(st *os.ProcessState, werr error) {
				// The CHILD's own code, read from ProcessState rather than
				// inferred from an error: the errgroup inside agentexec
				// surfaces whichever goroutine failed first, and a teardown
				// error routinely beats the exit status to it.
				if st != nil {
					exitCode = int32(st.ExitCode())
					kind = protocol.ExecEventKind_Exited
				}
				if werr != nil && st == nil {
					detail = werr.Error()
				}
			},
		})
	if kind == protocol.ExecEventKind_Failed && detail == "" && runErr != nil {
		detail = runErr.Error()
	}
	if kind == protocol.ExecEventKind_Failed && detail == "" {
		detail = fmt.Sprintf("exec %q did not start", argv[0])
	}
	finish(kind, exitCode, detail)
}
```

package server

import (
	"sort"
	"sync"
	"time"

	"github.com/on-keyday/agent-harness/runner/protocol"
	"github.com/on-keyday/objtrsf/trsf"
)

// execRun is one running out-of-band exec. The server holds it so `exec ls` can
// report it and `exec kill` can reach it; nothing here is persisted, because an
// exec dies with its caller (spec D10) and so cannot outlive the process that
// would replay a WAL.
type execRun struct {
	execID     uint64
	taskIDHex  string
	runnerID   string // = TaskEntry.AssignedTo; used to re-find the runner at kill
	argv       []string
	startedAt  time.Time
	control    trsf.BidirectionalStream
	cancel     func() // stops the runner side
	clientCID  string
	clientKind protocol.ClientKind
}

// execRegistry maps server-assigned execId → registration. Safe for concurrent
// use. Lives on TaskHandler beside portForwardRegistry.
type execRegistry struct {
	mu   sync.Mutex
	next uint64
	m    map[uint64]*execRun
}

func newExecRegistry() *execRegistry {
	return &execRegistry{m: map[uint64]*execRun{}}
}

// add assigns the next id and registers e. Ids start at 1 so that 0 is never a
// real one and a zero-valued field means "none".
func (r *execRegistry) add(e *execRun) uint64 {
	r.mu.Lock()
	defer r.mu.Unlock()
	r.next++
	e.execID = r.next
	if e.startedAt.IsZero() {
		e.startedAt = time.Now()
	}
	r.m[e.execID] = e
	return e.execID
}

func (r *execRegistry) get(id uint64) (*execRun, bool) {
	r.mu.Lock()
	defer r.mu.Unlock()
	e, ok := r.m[id]
	return e, ok
}

func (r *execRegistry) remove(id uint64) (*execRun, bool) {
	r.mu.Lock()
	defer r.mu.Unlock()
	e, ok := r.m[id]
	if ok {
		delete(r.m, id)
	}
	return e, ok
}

// list returns the registrations, ascending by id. taskFilter "" means all.
func (r *execRegistry) list(taskFilter string) []*execRun {
	r.mu.Lock()
	defer r.mu.Unlock()
	out := make([]*execRun, 0, len(r.m))
	for _, e := range r.m {
		if taskFilter != "" && e.taskIDHex != taskFilter {
			continue
		}
		out = append(out, e)
	}
	sort.Slice(out, func(i, j int) bool { return out[i].execID < out[j].execID })
	return out
}

// countForTask is what TaskInfo.exec_count reports. Returned as uint16 because
// that is the field's width; a task with more than 65535 concurrent execs has a
// different problem.
func (r *execRegistry) countForTask(taskIDHex string) uint16 {
	r.mu.Lock()
	defer r.mu.Unlock()
	var n uint16
	for _, e := range r.m {
		if e.taskIDHex == taskIDHex {
			n++
		}
	}
	return n
}

import (
	"encoding/hex"
	"log/slog"

	"github.com/on-keyday/agent-harness/appwire"
	"github.com/on-keyday/agent-harness/runner/protocol"

// handleOpenExecRun starts one out-of-band exec: allocate an id, create the
// data stream toward the client and the runner plus the control stream toward
// the client, tell the runner, register, answer.
//
// Same shape as handleGitQuery, with two differences: an exec needs a SECOND
// (control) stream toward the client, because its exit code cannot ride the
// exec frame protocol (spec D9); and it is registered, because it is long-lived
// enough to be listed and killed.
func (h *TaskHandler) handleOpenExecRun(conn ConnHandle, req *protocol.ExecRunRequest) protocol.ExecRunResponse {
	errResp := func(s protocol.ExecRunStatus) protocol.ExecRunResponse {
		return protocol.ExecRunResponse{Status: s}
	}
	taskIDHex := hex.EncodeToString(req.TaskId.Id[:])
	if req.Argv.ArgvLen == 0 {
		return errResp(protocol.ExecRunStatus_EmptyArgv)
	}
	task, ok := h.Tasks.Get(taskIDHex)
	if !ok {
		return errResp(protocol.ExecRunStatus_NotFound)
	}
	runner, ok := h.Registry.Get(task.AssignedTo)
	if !ok || runner.Conn == nil {
		return errResp(protocol.ExecRunStatus_RunnerUnreachable)
	}
	if conn == nil {
		slog.Error("exec_run: nil client conn (programmer error)")
		return errResp(protocol.ExecRunStatus_InternalError)
	}

	dataStream := conn.CreateBidirectionalStream()
	if dataStream == nil {
		return errResp(protocol.ExecRunStatus_InternalError)
	}
	ctrlStream := conn.CreateBidirectionalStream()
	if ctrlStream == nil {
		_ = dataStream.CloseBoth()
		return errResp(protocol.ExecRunStatus_InternalError)
	}
	runnerStream := runner.Conn.CreateBidirectionalStream()
	if runnerStream == nil {
		_ = dataStream.CloseBoth()
		_ = ctrlStream.CloseBoth()
		return errResp(protocol.ExecRunStatus_InternalError)
	}

	argv := make([]string, 0, req.Argv.ArgvLen)
	for i := range req.Argv.Argv {
		argv = append(argv, string(req.Argv.Argv[i].Arg))
	}
	e := &execRun{
		taskIDHex:  taskIDHex,
		runnerID:   task.AssignedTo.String(),
		argv:       argv,
		control:    ctrlStream,
		clientCID:  conn.ConnectionID().String(),
		clientKind: h.clientKindFor(conn),
	}
	execID := h.execs().add(e)

	rreq := protocol.RunnerRequest{Kind: protocol.RunnerRequestType_OpenExecRun}
	rreq.SetOpenExecRun(runnerExecRunRequest(req, execID, task.RepoPath, uint64(runnerStream.ID())))
	data := rreq.MustAppend([]byte{byte(appwire.AppKind_RunnerControl)})
	if _, _, err := runner.Conn.SendMessage(data); err != nil {
		h.execs().remove(execID)
		_ = dataStream.CloseBoth()
		_ = ctrlStream.CloseBoth()
		_ = runnerStream.CloseBoth()
		slog.Error("exec_run: send to runner failed", "task_id", taskIDHex, "err", err)
		return errResp(protocol.ExecRunStatus_InternalError)
	}
	go spliceBidi(dataStream, runnerStream, taskIDHex)

	return protocol.ExecRunResponse{
		Status:          protocol.ExecRunStatus_Ok,
		ExecId:          execID,
		DataStreamId:    uint64(dataStream.ID()),
		ControlStreamId: uint64(ctrlStream.ID()),
	}
}

// runnerExecRunRequest relays the client's request to the runner, adding the
// exec id, the stream id and the repo path the server holds.
//
// A separate function so a test can assert EVERY field survives: one relay site
// for a growing struct is the shape that has silently dropped a new field on
// this project before (runnerGitRequest carries the same note).
func runnerExecRunRequest(req *protocol.ExecRunRequest, execID uint64, repoPath string, streamID uint64) protocol.RunnerExecRunRequest {
	body := protocol.RunnerExecRunRequest{
		ExecId:   execID,
		TaskId:   req.TaskId,
		StreamId: streamID,
		Argv:     req.Argv,
	}
	body.SetRepoPath([]byte(repoPath))
	body.SetStdinEnabled(req.StdinEnabled())
	return body
}

// onExecRunFinished delivers the outcome to the waiting client and deregisters.
// An unknown id is a no-op: the client may have gone away first, taking the
// registration with it.
func (h *TaskHandler) onExecRunFinished(fin *protocol.ExecRunFinished) {
	e, ok := h.execs().remove(fin.ExecId)
	if !ok {
		return
	}
	ev := protocol.ExecEvent{Kind: fin.Kind, ExitCode: fin.ExitCode}
	ev.Detail = fin.Detail
	body, err := ev.Append(nil)
	if err != nil {
		slog.Error("exec_run: encode event", "exec_id", fin.ExecId, "err", err)
		_ = e.control.CloseBoth()
		return
	}
	if _, werr := e.control.Write(body); werr != nil {
		slog.Info("exec_run: client gone before the event landed", "exec_id", fin.ExecId)
	}
	_ = e.control.CloseBoth()
}

// handleExecRunKill stops one. Gated on exec_run against the TARGET TASK, not
// on who started it: the bit that authorizes running commands in a tree
// authorizes stopping them, and requiring `cancel` instead would leave a holder
// able to start what it cannot stop.
func (h *TaskHandler) handleExecRunKill(conn ConnHandle, req *protocol.ExecRunKillRequest) protocol.ExecRunKillResponse {
	e, ok := h.execs().get(req.ExecId)
	if !ok {
		return protocol.ExecRunKillResponse{Status: protocol.ExecRunStatus_NotFound}
	}
	if !h.callerMayTarget(conn, protocol.Capability_ExecRun, e.taskIDHex) {
		// Same shape as handleSubmitResume: an out-of-scope target is reported
		// as absent, because saying "denied" would confirm it exists.
		return protocol.ExecRunKillResponse{Status: protocol.ExecRunStatus_NotFound}
	}
	if e.cancel != nil {
		e.cancel()
	}
	h.execs().remove(req.ExecId)
	ev := protocol.ExecEvent{Kind: protocol.ExecEventKind_Killed, ExitCode: -1}
	ev.Detail = []byte("killed by exec kill")
	if body, err := ev.Append(nil); err == nil {
		_, _ = e.control.Write(body)
	}
	_ = e.control.CloseBoth()
	return protocol.ExecRunKillResponse{Status: protocol.ExecRunStatus_Ok}
}

import (

// ExecRunOpts carries the caller's three ends. A nil Stdin means the child's
// stdin is closed immediately, which is what a non-interactive caller wants.
type ExecRunOpts struct {
	Stdin  io.Reader
	Stdout io.Writer
	Stderr io.Writer
}

// ExecRunResult is how one exec ended.
type ExecRunResult struct {
	ExitCode int32
	Kind     protocol.ExecEventKind
	Detail   string
}

func buildExecArgv(argv []string) protocol.ExecArgv {
	var out protocol.ExecArgv
	for _, a := range argv {
		var e protocol.ExecArg
		e.SetArg([]byte(a))
		out.Argv = append(out.Argv, e)
	}
	out.ArgvLen = uint16(len(out.Argv))
	return out
}

// ExecRun runs argv in the task's worktree and blocks until it ends, copying
// stdout and stderr to opts as they arrive — SEPARATELY, which is the whole
// point of this verb versus `session exec`.
//
// The exit code arrives on a second stream, not in the frame protocol: that
// protocol has no exit shape and is pre-trsf legacy, so sideband goes in its
// own trsf stream (spec D9).
func (c *Client) ExecRun(ctx context.Context, taskIDHex string, argv []string, opts ExecRunOpts) (ExecRunResult, error) {
	tid, err := parseTaskIDHex(taskIDHex)
	if err != nil {
		return ExecRunResult{}, fmt.Errorf("exec: parse task id: %w", err)
	}
	if len(argv) == 0 {
		return ExecRunResult{}, errors.New("exec: empty command")
	}
	req := &protocol.TaskControlRequest{Kind: protocol.TaskControlKind_OpenExecRun}
	body := protocol.ExecRunRequest{TaskId: tid, Argv: buildExecArgv(argv)}
	body.SetStdinEnabled(opts.Stdin != nil)
	req.SetOpenExecRun(body)

	resp, err := c.RoundTripTaskControl(ctx, req)
	if err != nil {
		return ExecRunResult{}, err
	}
	r := resp.OpenExecRun()
	if r == nil {
		return ExecRunResult{}, errors.New("exec: response variant missing")
	}
	if err := execRunStatusError(taskIDHex, r.Status); err != nil {
		return ExecRunResult{}, err
	}

	data := peer.WaitForBidirectionalStream(ctx, c.Transport(), trsf.StreamID(r.DataStreamId))
	if data == nil {
		return ExecRunResult{}, fmt.Errorf("exec: data stream %d not visible", r.DataStreamId)
	}
	defer data.CloseBoth()
	ctrl := peer.WaitForBidirectionalStream(ctx, c.Transport(), trsf.StreamID(r.ControlStreamId))
	if ctrl == nil {
		return ExecRunResult{}, fmt.Errorf("exec: control stream %d not visible", r.ControlStreamId)
	}
	defer ctrl.CloseBoth()

	stream := agentexec.NewCommandExecutionStream(data)
	done := make(chan struct{}, 2)
	go func() { defer func() { done <- struct{}{} }(); _, _ = io.Copy(opts.Stdout, stream.Stdout()) }()
	go func() { defer func() { done <- struct{}{} }(); _, _ = io.Copy(opts.Stderr, stream.Stderr()) }()
	if opts.Stdin != nil {
		go func() { _, _ = io.Copy(stream.Stdin(), opts.Stdin) }()
	}

	raw, err := io.ReadAll(ctrl)
	if err != nil {
		return ExecRunResult{}, fmt.Errorf("exec: read outcome: %w", err)
	}
	<-done
	<-done

	var ev protocol.ExecEvent
	if err := ev.DecodeExact(raw); err != nil {
		return ExecRunResult{}, fmt.Errorf("exec: decode outcome: %w", err)
	}
	return ExecRunResult{ExitCode: ev.ExitCode, Kind: ev.Kind, Detail: string(ev.Detail)}, nil
}

func execRunStatusError(taskID string, st protocol.ExecRunStatus) error {
	switch st {
	case protocol.ExecRunStatus_Ok:
		return nil
	case protocol.ExecRunStatus_NotFound:
		return fmt.Errorf("exec: task %q not found (pruned, wrong id, or out of your scope)", taskID)
	case protocol.ExecRunStatus_NoWorktree:
		return fmt.Errorf("exec: task %q has no worktree to run in — it ended clean and its tree was removed", taskID)
	case protocol.ExecRunStatus_RunnerUnreachable:
		return fmt.Errorf("exec: the runner hosting task %q is not connected", taskID)
	case protocol.ExecRunStatus_EmptyArgv:
		return errors.New("exec: empty command")
	case protocol.ExecRunStatus_Denied:
		return errors.New("exec: denied (needs the exec_run capability)")
	default:
		return fmt.Errorf("exec: %s", st.String())
	}
}
